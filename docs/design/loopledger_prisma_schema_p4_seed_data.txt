# LoopLedger — Prisma Schema & P4 Seed Data Test Suite

This package provides the complete `prisma/schema.prisma` and an executable `prisma/seed.ts` specifically tailored for **P4 (Money, Reporting & Portal Lead)** to verify:
1. **Revenue Radar Calculations**: MRR, ARR, Collected vs. Outstanding balance aggregation.
2. **Ageing Buckets**: Exact test fixtures for `0–30 Days`, `31–60 Days`, and `60+ Days` overdue delinquency buckets.
3. **Customer Health Badges**: Test scenarios for `GREEN`, `AMBER`, and `RED` critical status.
4. **Customer Portal Scopes**: Multi-tenant customer isolation (`customer_id` filter) and self-service subscriptions.
5. **Idempotency & Partial Payments**: Invoices with partial payment balances and UPI settlement test records.

## 1. Complete Prisma Schema (`prisma/schema.prisma`)

```prisma
datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

enum Role {
  ADMIN
  STAFF
  CUSTOMER
}

enum BillingPeriod {
  DAILY
  WEEKLY
  MONTHLY
  YEARLY
}

enum SubscriptionStatus {
  DRAFT
  QUOTATION
  CONFIRMED
  ACTIVE
  CLOSED
}

enum InvoiceStatus {
  DRAFT
  CONFIRMED
  PAID
  CANCELLED
}

enum PaymentMethod {
  CASH
  CARD
  UPI
  BANK
}

enum DiscountType {
  FIXED
  PERCENT
}

enum DiscountAppliesTo {
  PRODUCT
  SUBSCRIPTION
}

// ----------------------------------------------------
// 1. Users & Multitenant Customers
// ----------------------------------------------------

model User {
  id            String    @id @default(uuid())
  email         String    @unique
  password_hash String
  role          Role      @default(STAFF)
  customer_id   String?
  customer      Customer? @relation(fields: [customer_id], references: [id], onDelete: Cascade)
  created_at    DateTime  @default(now())
  updated_at    DateTime  @updatedAt

  @@index([role])
  @@index([customer_id])
}

model Customer {
  id            String         @id @default(uuid())
  name          String
  company       String
  phone         String
  address       String
  gstin         String?
  created_at    DateTime       @default(now())
  updated_at    DateTime       @updatedAt

  users         User[]
  subscriptions Subscription[]
  invoices      Invoice[]

  @@index([company])
}

// ----------------------------------------------------
// 2. Products, Variants, & Catalog
// ----------------------------------------------------

model Product {
  id                 String             @id @default(uuid())
  name               String
  type               String             // e.g. "Core Platform", "Add-on", "SLA Pack"
  sales_price        Float              // Must be >= 0
  cost_price         Float              // Must be >= 0
  notes              String?
  active             Boolean            @default(true)
  created_at         DateTime           @default(now())

  variants           ProductVariant[]
  template_lines     TemplateLine[]
  subscription_lines SubscriptionLine[]
}

model ProductVariant {
  id                 String             @id @default(uuid())
  product_id         String
  product            Product            @relation(fields: [product_id], references: [id], onDelete: Cascade)
  attribute          String             // e.g. "Connector / Brand", "Deployment Node"
  value              String             // e.g. "Odoo", "SAP ERP Bridge", "Dedicated Mumbai VPC"
  extra_price        Float              // e.g. 560.00
  active             Boolean            @default(true)

  template_lines     TemplateLine[]
  subscription_lines SubscriptionLine[]

  @@unique([product_id, attribute, value])
  @@index([product_id])
}

// ----------------------------------------------------
// 3. Plans, Quotations & Templates
// ----------------------------------------------------

model Plan {
  id             String         @id @default(uuid())
  name           String
  price          Float
  billing_period BillingPeriod  @default(MONTHLY)
  min_qty        Int            @default(1)
  start_date     DateTime
  end_date       DateTime?
  auto_close     Boolean        @default(false)
  closable       Boolean        @default(true)
  pausable       Boolean        @default(true)
  renewable      Boolean        @default(true)

  templates      QuotationTemplate[]
  subscriptions  Subscription[]
}

model QuotationTemplate {
  id            String         @id @default(uuid())
  name          String
  validity_days Int            @default(15)
  plan_id       String
  plan          Plan           @relation(fields: [plan_id], references: [id])
  lines         TemplateLine[]
}

model TemplateLine {
  id          String          @id @default(uuid())
  template_id String
  template    QuotationTemplate @relation(fields: [template_id], references: [id], onDelete: Cascade)
  product_id  String
  product     Product         @relation(fields: [product_id], references: [id])
  variant_id  String?
  variant     ProductVariant? @relation(fields: [variant_id], references: [id])
  qty         Int             @default(1)
}

// ----------------------------------------------------
// 4. Subscriptions & Line Items
// ----------------------------------------------------

model Subscription {
  id                String             @id @default(uuid())
  number            String             @unique // e.g. "SUB-0284"
  customer_id       String
  customer          Customer           @relation(fields: [customer_id], references: [id])
  plan_id           String
  plan              Plan               @relation(fields: [plan_id], references: [id])
  start_date        DateTime
  expiry_date       DateTime?
  payment_term      String             @default("NET_30") // e.g. "IMMEDIATE", "NET_15", "NET_30"
  status            SubscriptionStatus @default(DRAFT)
  next_invoice_date DateTime
  created_at        DateTime           @default(now())
  updated_at        DateTime           @updatedAt

  lines             SubscriptionLine[]
  invoices          Invoice[]

  @@index([status])
  @@index([customer_id])
  @@index([next_invoice_date])
}

model SubscriptionLine {
  id              String          @id @default(uuid())
  subscription_id String
  subscription    Subscription    @relation(fields: [subscription_id], references: [id], onDelete: Cascade)
  product_id      String
  product         Product         @relation(fields: [product_id], references: [id])
  variant_id      String?
  variant         ProductVariant? @relation(fields: [variant_id], references: [id])
  qty             Int             @default(1)
  unit_price      Float
  tax_id          String?
  tax             Tax?            @relation(fields: [tax_id], references: [id])
  amount          Float

  @@index([subscription_id])
}

// ----------------------------------------------------
// 5. Taxes & Discounts
// ----------------------------------------------------

model Tax {
  id                 String             @id @default(uuid())
  name               String             // e.g. "GST 18%", "GST 12%"
  rate_percent       Float              // 0 to 100
  active             Boolean            @default(true)
  subscription_lines SubscriptionLine[]
}

model Discount {
  id           String            @id @default(uuid())
  name         String            // e.g. "DI-ENTERPRISE-Q4", "DI-FESTIVE25"
  type         DiscountType      @default(PERCENT)
  fixed_amount Float?            // in Rupees
  percent      Float?            // whole number (e.g. 10.0 for 10%)
  min_purchase Float?
  min_qty      Int?
  start_date   DateTime
  end_date     DateTime
  usage_limit  Int               @default(100)
  used_count   Int               @default(0)
  applies_to   DiscountAppliesTo @default(SUBSCRIPTION)
  target_id    String?           // product or plan ID if applicable
}

// ----------------------------------------------------
// 6. Invoices & Payments (P4 Core Domain)
// ----------------------------------------------------

model Invoice {
  id              String        @id @default(uuid())
  number          String        @unique // e.g. "INV-2026-0412"
  subscription_id String
  subscription    Subscription  @relation(fields: [subscription_id], references: [id])
  customer_id     String
  customer        Customer      @relation(fields: [customer_id], references: [id])
  period_start    DateTime
  period_end      DateTime
  status          InvoiceStatus @default(CONFIRMED)
  subtotal        Float
  discount_total  Float         @default(0)
  tax_total       Float         @default(0)
  total           Float
  due_date        DateTime
  created_at      DateTime      @default(now())
  updated_at      DateTime      @updatedAt

  lines           InvoiceLine[]
  payments        Payment[]

  // Critical spec constraint preventing duplicate billing runs:
  @@unique([subscription_id, period_start])
  @@index([status])
  @@index([customer_id])
  @@index([due_date])
}

model InvoiceLine {
  id              String   @id @default(uuid())
  invoice_id      String
  invoice         Invoice  @relation(fields: [invoice_id], references: [id], onDelete: Cascade)
  description     String
  qty             Int      @default(1)
  unit_price      Float
  tax_amount      Float    @default(0)
  discount_amount Float    @default(0)
  amount          Float

  @@index([invoice_id])
}

model Payment {
  id          String        @id @default(uuid())
  invoice_id  String
  invoice     Invoice       @relation(fields: [invoice_id], references: [id], onDelete: Cascade)
  method      PaymentMethod @default(UPI)
  amount      Float         // Must be > 0 and sum <= invoice.total
  paid_on     DateTime      @default(now())
  note        String?
  recorded_by String        // user ID or 'SYSTEM_UPI_AUTONOMOUS'
  created_at  DateTime      @default(now())

  @@index([invoice_id])
  @@index([paid_on])
}

// ----------------------------------------------------
// 7. Audit Log
// ----------------------------------------------------

model AuditLog {
  id        String   @id @default(uuid())
  entity    String   // e.g. "INVOICE", "PAYMENT", "SUBSCRIPTION"
  entity_id String
  action    String   // e.g. "RECORD_PAYMENT", "STATUS_CHANGE", "DISCOUNT_APPLIED"
  user_id   String
  at        DateTime @default(now())
  detail    String   // JSON stringified metadata

  @@index([entity, entity_id])
  @@index([at])
}
```

## 3. How to Run in VS Code

1. In your `loopledger` directory, update `prisma/schema.prisma` with the content above.
2. Ensure `package.json` includes the Prisma seed configuration:
   ```json
   "prisma": {
     "seed": "tsx prisma/seed.ts"
   }
   ```
3. Run migrations and execute the seed:
   ```bash
   pnpm prisma migrate dev --name init_p4_schema
   pnpm prisma db seed
   ```
4. Verify by running the reports pure functions in `src/server/reports.ts` or launching `pnpm dev` and visiting `/dashboard` and `/my-invoices`.
